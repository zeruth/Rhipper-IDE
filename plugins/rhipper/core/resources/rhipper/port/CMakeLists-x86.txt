# The CMake project of the port of @NAME@. Rhipper IDE wrote this file because the project had none.
# The IDE does not write it again, so you can change it.
cmake_minimum_required(VERSION 3.24)
project(@TARGET@ C CXX)

# The port is an x64 program that runs the translation of the x86 reference.
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "the port builds as x64 (-A x64)")
endif()
set(RHIPPER_DIR "@RHIPPER_DIR@" CACHE PATH "The Rhipper directory")
set(RUNTIME32 ${RHIPPER_DIR}/runtime32)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

# The translation that translate32.py writes. A call to an undeclared function is an error (C4013).
file(GLOB GEN_SOURCES CONFIGURE_DEPENDS "@GEN_DIR@/*.c")
add_library(@GEN_TARGET@ OBJECT ${GEN_SOURCES})
target_include_directories(@GEN_TARGET@ PRIVATE "@GEN_DIR@")
target_compile_options(@GEN_TARGET@ PRIVATE /W1 /bigobj /O2 /GS- /TC /Zi
  /we4013 /wd4022 /wd4024 /wd4028 /wd4047 /wd4090 /wd4133 /wd4146 /wd4244 /wd4305 /wd4311 /wd4312
  /wd4554 /wd4700 /wd4715 /wd4716)

include(${RUNTIME32}/Runtime32.cmake)
add_executable(@TARGET@ $<TARGET_OBJECTS:@GEN_TARGET@>)
rhipper_runtime32(@TARGET@ NAME "@NAME@" ENV "@ENV@" REF "@REF@")
# No identical-code folding: two folded adapters of one body get one name in the PDB.
target_link_options(@TARGET@ PRIVATE /DEBUG:FULL /OPT:REF /OPT:NOICF)
